Key Takeaway (In Plain English)

SQL Aggregations & Python Z-Score Anomaly Detection Logic 
-- SQL: Hourly Diurnal Load & Thermal Profile (from sql/telemetry_aggregations.sql)
SELECT
    CAST(strftime('%H', timestamp) AS INTEGER) AS hour_of_day,
    ROUND(AVG(temperature_c), 2) AS avg_temp_c,
    ROUND(AVG(power_kw), 3) AS avg_power_kw
FROM telemetry
GROUP BY hour_of_day
ORDER BY hour_of_day;
